import Link from "next/link";
import { notFound } from "next/navigation";

import { requireAnyRole } from "@/lib/auth/permissions";
import { EDITORIAL_ROLES } from "@/lib/auth/role-groups";
import { prisma } from "@/lib/prisma";

import { sendEditorialDecision } from "./actions";

type PageProps = {
  params: Promise<{
    id: string;
  }>;
};


function formatStatus(value: string) {
  return value
    .replaceAll("_", " ")
    .toLowerCase()
    .replace(/\b\w/g, (character) =>
      character.toUpperCase(),
    );
}

function formatDate(
  value: Date | null | undefined,
) {
  if (!value) {
    return "Not available";
  }

  return new Intl.DateTimeFormat("en-GB", {
    day: "numeric",
    month: "long",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  }).format(value);
}

function recommendationStyle(
  recommendation: string | null | undefined,
) {
  switch (recommendation) {
    case "ACCEPT":
      return "bg-green-100 text-green-800";

    case "MINOR_REVISION":
      return "bg-yellow-100 text-yellow-800";

    case "MAJOR_REVISION":
      return "bg-orange-100 text-orange-800";

    case "REJECT":
      return "bg-red-100 text-red-800";

    default:
      return "bg-gray-100 text-gray-700";
  }
}

export default async function EditorialDecisionPage({
  params,
}: PageProps) {
  const editor =
    await requireAnyRole(EDITORIAL_ROLES);

  const { id } = await params;

  const submission =
    await prisma.submission.findUnique({
      where: {
        id,
      },
      include: {
        author: true,
        journal: true,

        versions: {
          orderBy: {
            versionNumber: "desc",
          },
          take: 1,
        },

        reviewAssignments: {
          include: {
            reviewer: {
              select: {
                id: true,
                name: true,
                email: true,
              },
            },

            review: true,
            submissionVersion: true,
          },

          orderBy: [
            {
              reviewRound: "desc",
            },
            {
              invitedAt: "asc",
            },
          ],
        },

        decisions: {
          include: {
            editor: true,
          },

          orderBy: {
            createdAt: "desc",
          },
        },
      },
    });

  if (!submission) {
    notFound();
  }

  const editorJournalRole =
    await prisma.userJournalRole.findFirst({
      where: {
        userId: editor.id,
        journalId: submission.journalId,
        role: {
          in: EDITORIAL_ROLES,
        },
      },
      select: {
        id: true,
      },
    });

  if (!editorJournalRole) {
    notFound();
  }

  const currentRound =
    submission.currentReviewRound || 1;

  const currentRoundAssignments =
    submission.reviewAssignments.filter(
      (assignment) =>
        assignment.reviewRound === currentRound,
    );

  const submittedAssignments =
    currentRoundAssignments.filter(
      (assignment) =>
        assignment.status === "SUBMITTED" &&
        assignment.review,
    );

    currentRoundAssignments.filter(
      (assignment) =>
        assignment.status === "INVITED" ||
        assignment.status === "ACCEPTED",
    );

    currentRoundAssignments.filter(
      (assignment) =>
        assignment.status === "DECLINED",
    );

  const roundComplete =
    currentRoundAssignments.length > 0 &&
    pendingAssignments.length === 0 &&
    submittedAssignments.length > 0;

  const latestVersion =
    submission.versions[0];

  const terminalStatus = [
    "ACCEPTED",

    "PUBLISHED",
    "WITHDRAWN",

        "MINOR_REVISION",
    ).length,

      (assignment) =>
        assignment.review?.recommendation ===
        "MAJOR_REVISION",
    ).length,

    REJECT: submittedAssignments.filter(
      (assignment) =>
        assignment.review?.recommendation ===
        "REJECT",
    ).length,
  };

      <header>
        <Link
          href={`/editor/submissions/${submission.id}`}
          className="text-sm font-medium text-blue-700 hover:underline"

          ← Back to editor workspace
        </Link>

        <div className="mt-4 flex flex-wrap items-start justify-between gap-4">
          <div>
            <h1 className="text-3xl font-bold">
              Editorial Decision
            </h1>

            <p className="mt-2 text-gray-600">
              Assess the reports submitted during review
              round {currentRound} and issue the next
              editorial decision.
            </p>
          </div>

            <Link
              href={`/editor/submissions/${submission.id}/review-rounds`}
              className="rounded-lg bg-cyan-700 px-5 py-3 font-semibold text-white hover:bg-cyan-800"
            >
              Review-Round Summary

                  defaultValue=""

                  </option>
                </select>

              <div>
                <label

                  Revision Deadline

                  revision.
                </p>

                <input
                  id="revisionDeadline"
                  name="revisionDeadline"
                  type="date"
                  className="w-full rounded-lg border p-3"

              </div>

              <div>
                <label
                  htmlFor="comments"
                  className="mb-2 block font-semibold"
                >
                  Decision Letter / Comments to Author

                <textarea
                  id="comments"
                  name="comments"

                  required
                  className="w-full rounded-lg border p-3"
                  placeholder="Dear Author, after considering the reviewer reports from the current review round..."

          </section>
        )}

      <section className="rounded-xl border bg-white p-6 shadow-sm">
        <h2 className="text-xl font-semibold">

          <p className="mt-4 text-gray-600">
            No editorial decision has been recorded.

          <div className="mt-5 space-y-4">
            {submission.decisions.map(
              (previousDecision) => (
                <article
                  key={previousDecision.id}
                  className="rounded-lg border p-4"

                  <div className="flex flex-wrap justify-between gap-3">

                        {formatStatus(
                          previousDecision.decision,

                      <p className="mt-1 text-sm text-gray-600">
                        By{" "}
                        {

                        on{" "}
                        {formatDate(
                          previousDecision.createdAt,

                      </p>
                    </div>

                    {previousDecision.revisionDeadline && (
                      <p className="text-sm font-medium">
                        Revision deadline:{" "}
                        {formatDate(
                          previousDecision.revisionDeadline,

                      </p>

                  <div className="mt-4 whitespace-pre-wrap rounded-lg bg-gray-50 p-4 text-sm">
                    {previousDecision.comments ||
                      "No decision comments were recorded."}
                  </div>

              ),
            )}

        )}
      </section>
    </main>
  );

===== EDITOR DECISION ACTIONS =====
"use server";

import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

import { requireAnyActionRole } from "@/lib/auth/permissions";
import { EDITORIAL_ROLES } from "@/lib/auth/role-groups";
import { sendEmail } from "@/lib/email/send-email";
import {
  manuscriptAcceptedEmail,
  manuscriptRejectedEmail,
  revisionRequestedEmail,
} from "@/lib/email/templates";
import { prisma } from "@/lib/prisma";

const VALID_DECISIONS = [
  "REVISION_REQUIRED",
  "ACCEPTED",
  "REJECTED",
] as const;

type EditorialDecisionValue =
  (typeof VALID_DECISIONS)[number];

const EDITOR_ROLES = [
  "SUPER_ADMIN",
  "ADMIN",
  "EDITOR_IN_CHIEF",
  "MANAGING_EDITOR",
  "SECTION_EDITOR",
] as const;

function getAppUrl() {
  return (
    process.env.NEXT_PUBLIC_APP_URL ||
    "http://localhost:3000"
  ).replace(/\/+$/, "");
}

async function deliverDecisionEmail({
  to,
  submissionId,
  decision,
  emailContent,
}: {
  to: string;
  submissionId: string;
  decision: EditorialDecisionValue;
  emailContent: {
    subject: string;
    html: string;
    text: string;
  };
}) {
  try {
    await sendEmail({
      to,
      subject: emailContent.subject,
      html: emailContent.html,
      text: emailContent.text,
    });
  } catch (error) {
    console.error(
      "Editorial decision email delivery failed:",
      {
        submissionId,
        decision,
        recipient: to,
        error,
      },
    );
  }
}

export async function sendEditorialDecision(
  formData: FormData,
) {
  const editor =
    await requireAnyActionRole(
      EDITORIAL_ROLES,
    );

  const submissionId = String(
    formData.get("submissionId") || "",
  ).trim();

  const decision = String(
    formData.get("decision") || "",
  ).trim() as EditorialDecisionValue;

  const comments = String(
    formData.get("comments") || "",
  ).trim();

  const revisionDeadlineValue = String(
    formData.get("revisionDeadline") || "",
  ).trim();

  if (!submissionId) {
    throw new Error(
      "Submission ID is missing.",
    );
  }

  if (!VALID_DECISIONS.includes(decision)) {
    throw new Error(
      "Please select a valid editorial decision.",
    );
  }

  if (!comments) {
    throw new Error(
      "A decision letter or comments to the author are required.",
    );
  }

  let revisionDeadline: Date | null = null;

  if (decision === "REVISION_REQUIRED") {
    if (!revisionDeadlineValue) {
      throw new Error(
        "A revision deadline is required when requesting a revision.",
      );
    }

    revisionDeadline = new Date(
      `${revisionDeadlineValue}T23:59:59`,
    );

    if (
      Number.isNaN(
        revisionDeadline.getTime(),
      )
    ) {
      throw new Error(
        "The revision deadline is invalid.",
      );
    }

    if (revisionDeadline <= new Date()) {
      throw new Error(
        "The revision deadline must be in the future.",
      );
    }
  }

  const submission =
    await prisma.submission.findUnique({
      where: {
        id: submissionId,
      },

      include: {
        author: {
          select: {
            id: true,
            name: true,
            email: true,
          },
        },

        journal: {
          select: {
            id: true,
            name: true,
            shortName: true,
          },
        },
      },
    });

  if (!submission) {
    throw new Error(
      "Submission not found.",
    );
  }

  const terminalStatuses = [
    "ACCEPTED",
    "REJECTED",
    "PUBLISHED",
    "WITHDRAWN",
  ] as const;

  if (
    terminalStatuses.includes(
      submission.status as
      (typeof terminalStatuses)[number],
    )
  ) {
    throw new Error(
      `A new decision cannot be recorded because this submission is already ${submission.status
        .replaceAll("_", " ")
        .toLowerCase()}.`,
    );
  }

  const editorJournalRole =
    await prisma.userJournalRole.findFirst({
      where: {
        userId: editor.id,
        journalId: submission.journalId,

        role: {
          in: [...EDITOR_ROLES],
        },
      },

      select: {
        id: true,
      },
    });

  if (!editorJournalRole) {
    throw new Error(
      "You do not have editorial permission for this journal.",
    );
  }

  if (!editor.isActive) {
    throw new Error(
      "This editor account is inactive.",
    );
  }

  const editorHasPermission =
    editor.roles.some((roleRecord) =>
      EDITOR_ROLES.includes(
        roleRecord.role as
        (typeof EDITOR_ROLES)[number],
      ),
    );

  if (!editorHasPermission) {
    throw new Error(
      "Your account does not have permission to issue editorial decisions.",
    );
  }

  const currentRound =
    submission.currentReviewRound || 1;

  const currentRoundAssignments =
    await prisma.reviewAssignment.findMany({
      where: {
        submissionId,
        reviewRound: currentRound,
      },

      include: {
        review: true,
      },
    });

  if (currentRoundAssignments.length === 0) {
    throw new Error(
      `No reviewer assignments were found for review round ${currentRound}.`,
    );
  }

  const pendingAssignments =
    currentRoundAssignments.filter(
      (assignment) =>
        assignment.status === "INVITED" ||
        assignment.status === "ACCEPTED",
    );

  if (pendingAssignments.length > 0) {
    throw new Error(
      `An editorial decision cannot yet be made. ${pendingAssignments.length
      } review assignment${pendingAssignments.length === 1
        ? " is"
        : "s are"
      } still outstanding in review round ${currentRound}.`,
    );
  }

  const submittedReviews =
    currentRoundAssignments.filter(
      (assignment) =>
        assignment.status === "SUBMITTED" &&
        assignment.review,
    );

"use server";

import { randomUUID } from "crypto";
import { mkdir, writeFile } from "fs/promises";
import path from "path";
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

import { requireAnyActionRole } from "@/lib/auth/permissions";

const MAX_FILE_SIZE = 20 * 1024 * 1024;

const ALLOWED_FILE_EXTENSIONS = [
  ".pdf",
  ".doc",
  ".docx",
];

function sanitizeFileName(fileName: string) {
  return fileName
    .replace(/[^a-zA-Z0-9._-]/g, "-")
    .replace(/-+/g, "-");
}

function validateUploadedFile(file: File, required: boolean) {
  if (!file || file.size === 0) {
    if (required) {
      throw new Error("The revised manuscript file is required.");
    }

    return;
  }

  if (file.size > MAX_FILE_SIZE) {
    throw new Error(
      `${file.name} is larger than the permitted 20 MB file size.`,
    );
  }

  const extension = path.extname(file.name).toLowerCase();

  if (!ALLOWED_FILE_EXTENSIONS.includes(extension)) {
    throw new Error(
      `${file.name} is not supported. Upload a PDF, DOC, or DOCX file.`,
    );
  }
}

async function saveUploadedFile(
  file: File,
  submissionId: string,
  fileCategory: "manuscript" | "response",
) {
  validateUploadedFile(file, fileCategory === "manuscript");

  const extension = path.extname(file.name).toLowerCase();
  const originalName = sanitizeFileName(file.name);
  const storedName = `${fileCategory}-${randomUUID()}${extension}`;

  const relativeDirectory = path.join(
    "uploads",
    "revisions",
    submissionId,
  );

  const absoluteDirectory = path.join(
    process.cwd(),
    "public",
    relativeDirectory,
  );

  await mkdir(absoluteDirectory, {
    recursive: true,
  });

  const bytes = await file.arrayBuffer();
  const buffer = Buffer.from(bytes);

  await writeFile(
    path.join(absoluteDirectory, storedName),
    buffer,
  );

  return {
    originalName,
    publicUrl: `/${relativeDirectory
      .replaceAll("\\", "/")}/${storedName}`,
  };
}

export async function submitRevision(formData: FormData) {
  const author = await requireAnyActionRole(AUTHOR_ROLES);

  const submissionId = String(
    formData.get("submissionId") || "",
  );

  const authorNotes = String(
    formData.get("authorNotes") || "",
  ).trim();

  const coverLetter = String(
    formData.get("coverLetter") || "",
  ).trim();

  const manuscriptFile = formData.get(
    "manuscriptFile",
  ) as File | null;

  const responseFile = formData.get(
    "responseFile",
  ) as File | null;

  if (!submissionId) {
    throw new Error("Submission ID is missing.");
  }

  if (!manuscriptFile) {
    throw new Error("Revised manuscript file is required.");
  }

  validateUploadedFile(manuscriptFile, true);

  if (responseFile && responseFile.size > 0) {
    validateUploadedFile(responseFile, false);
  }

  const submission = await prisma.submission.findFirst({
    where: {
      id: submissionId,
      authorId: author.id,
    },
    include: {
      versions: {
        orderBy: {
          versionNumber: "desc",
        },
        take: 1,
      },
      decisions: {
        orderBy: {
          createdAt: "desc",
        },
        take: 1,
      },
    },
  });

  if (!submission) {
    throw new Error(
      "Submission not found or you do not have permission to revise it.",
    );
  }

  if (submission.status !== "REVISION_REQUIRED") {
    throw new Error(
      "A revision cannot be uploaded because this submission is not currently awaiting revision.",
    );
  }

  const latestDecision = submission.decisions[0];

  if (
    latestDecision?.revisionDeadline &&
    new Date() > latestDecision.revisionDeadline
  ) {
    throw new Error(
      "The revision deadline has passed. Contact the editorial office for assistance.",
    );
  }

  const manuscriptUpload = await saveUploadedFile(
    manuscriptFile,
    submissionId,
    "manuscript",
  );

  let responseUpload:
    | {
      originalName: string;
      publicUrl: string;
    }
    | undefined;

  if (responseFile && responseFile.size > 0) {
    responseUpload = await saveUploadedFile(
      responseFile,
      submissionId,
      "response",
    );
  }

  const latestVersionNumber =
    submission.versions[0]?.versionNumber || 1;

  const nextVersionNumber = latestVersionNumber + 1;

  await prisma.$transaction(async (tx) => {
    await tx.submissionVersion.create({
      data: {
        submissionId,
        uploadedById: author.id,
        versionNumber: nextVersionNumber,
        manuscriptFileUrl: manuscriptUpload.publicUrl,
        manuscriptFileName: manuscriptUpload.originalName,
        responseFileUrl: responseUpload?.publicUrl,
        responseFileName: responseUpload?.originalName,
        coverLetter: coverLetter || null,
        authorNotes: authorNotes || null,
      },
    });

    await tx.submission.update({
      where: {
        id: submissionId,
      },
      data: {
        status: "UNDER_INITIAL_REVIEW",
      },
    });

    if (latestDecision?.editorId) {
      await tx.notification.create({
        data: {
          userId: latestDecision.editorId,
          type: "SUBMISSION",
          title: "Revised manuscript submitted",
          message: `The author submitted Version ${nextVersionNumber} of: ${submission.title}`,
          link: `/editor/submissions/${submissionId}`,
        },
      });
    }

    await tx.activityLog.create({
      data: {
        userId: author.id,
        action: "UPDATE",
        entity: "SubmissionVersion",
        entityId: submissionId,
        metadata: {
          versionNumber: nextVersionNumber,
          manuscriptFileName: manuscriptUpload.originalName,
          responseFileName:
            responseUpload?.originalName || null,
        },
      },
    });
  });

  revalidatePath(
    `/author/submissions/${submissionId}/revision`,
  );

  revalidatePath(
    `/author/submissions/${submissionId}`,
  );

  revalidatePath("/author/dashboard");
  revalidatePath(`/editor/submissions/${submissionId}`);
  revalidatePath("/editor/dashboard");

  redirect(`/author/submissions/${submissionId}`);
}

"use server";

import { revalidatePath } from "next/cache";

const REVIEW_PERIOD_DAYS = 14;

async function verifyEditorialPermission(
  userId: string,
  journalId: string,
) {
  const role = await prisma.userJournalRole.findFirst({
    where: {
      userId,
      journalId,
      role: {
        in: EDITORIAL_ROLES,
      },
    },

      id: true,
    },
  });

  if (!role) {
    throw new Error(
      "You do not have editorial permission for this journal.",
    );
  }
}

export async function sendRevisionToReviewers(
  formData: FormData,
) {
  const editor = await requireAnyActionRole(EDITORIAL_ROLES);

  const submissionId = String(
    formData.get("submissionId") || "",
  ).trim();

  if (!submissionId) {
    throw new Error("Submission ID is missing.");
  }

  const submission = await prisma.submission.findUnique({
    where: {
      id: submissionId,
    },
    include: {
      versions: {
        orderBy: {
          versionNumber: "desc",
        },
        take: 1,
      },

      reviewAssignments: {
        orderBy: [
          {
            reviewRound: "desc",
          },
          {
            createdAt: "asc",
          },
        ],
        include: {
          reviewer: {
            select: {
              id: true,
              name: true,
              email: true,
              isActive: true,
            },
          },
        },
      },
    },
  });

  if (!submission) {
    throw new Error("Submission not found.");
  }

  await verifyEditorialPermission(
    editor.id,
    submission.journalId,
  );

  if (
    submission.status === "ACCEPTED" ||
    submission.status === "REJECTED" ||
    submission.status === "PUBLISHED" ||
    submission.status === "WITHDRAWN"
  ) {
    throw new Error(
      `This submission cannot be sent for re-review because its current status is ${submission.status
        .replaceAll("_", " ")
        .toLowerCase()}.`,
    );
  }

  const latestVersion = submission.versions[0];

  if (!latestVersion) {
    throw new Error(
      "No revised manuscript version was found. The author must upload a revision before it can be sent to reviewers.",
    );
  }

  const previousRound =
    submission.currentReviewRound || 1;

  const nextRound = previousRound + 1;

  const previousRoundAssignments =
    submission.reviewAssignments.filter(
      (assignment) =>
        assignment.reviewRound === previousRound,
    );

  if (previousRoundAssignments.length === 0) {
    throw new Error(
      `No reviewers were found for review round ${previousRound}. Assign reviewers before starting the next round.`,
    );
  }

  const activeReviewers =
    previousRoundAssignments.filter(
      (assignment) =>
        assignment.reviewer.isActive,
    );

  if (activeReviewers.length === 0) {
    throw new Error(
      "None of the previous reviewers currently has an active account.",
    );
  }

  const existingNextRoundAssignments =
    await prisma.reviewAssignment.count({
      where: {
        submissionId,
        reviewRound: nextRound,
      },
    });

  if (existingNextRoundAssignments > 0) {
    throw new Error(
      `Review round ${nextRound} has already been created for this submission.`,
    );
  }

  const invitedAt = new Date();
  const dueDate = new Date(invitedAt);

  dueDate.setDate(
    dueDate.getDate() + REVIEW_PERIOD_DAYS,
  );

  await prisma.$transaction(async (tx) => {
    for (const previousAssignment of activeReviewers) {
      const newAssignment =
        await tx.reviewAssignment.create({
          data: {
            submissionId,
            reviewerId:
              previousAssignment.reviewerId,
            submissionVersionId:
              latestVersion.id,
            reviewRound: nextRound,
            status: "INVITED",
            invitedAt,
            dueDate,
            blindType:
              previousAssignment.blindType ||
              "DOUBLE_BLIND",
          },
        });

      await tx.notification.create({
        data: {
          userId:
            previousAssignment.reviewerId,
          type: "REVIEW",
          title: `Re-review invitation — Round ${nextRound}`,
          message:
            `A revised version of "${submission.title}" ` +
            `has been submitted. You are invited to review ` +
            `Version ${latestVersion.versionNumber} in review round ${nextRound}.`,
          link: `/reviewer/reviews/${newAssignment.id}`,
        },
      });
    }

    await tx.submission.update({
      where: {
        id: submissionId,
      },
      data: {
        currentReviewRound: nextRound,
        status: "UNDER_PEER_REVIEW",
      },
    });

    await tx.activityLog.create({
      data: {
        userId: editor.id,
        action: "ASSIGN_REVIEWER",
        entity: "Submission",
        entityId: submissionId,
        metadata: {
          event: "SEND_REVISION_TO_REVIEWERS",
          previousRound,
          newRound: nextRound,
          versionId: latestVersion.id,
          versionNumber:
            latestVersion.versionNumber,
          reviewerCount:
            activeReviewers.length,
          reviewerIds: activeReviewers.map(
            (assignment) =>
              assignment.reviewerId,
          ),
          dueDate: dueDate.toISOString(),
        },
      },
    });
  });

  revalidatePath(
    `/editor/submissions/${submissionId}`,
  );

  revalidatePath(
    `/editor/submissions/${submissionId}/revisions`,
  );

  revalidatePath(
    `/editor/submissions/${submissionId}/reviewers`,
  );

  revalidatePath("/editor/dashboard");
  revalidatePath("/reviewer/dashboard");

  redirect(
    `/editor/submissions/${submissionId}/reviewers`,
  );
}
